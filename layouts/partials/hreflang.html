{{- /* Generate hreflang links for all language versions of the current page. Shared prompts/tags/models: Hugo generates separate HTML per language but .Translations is empty. Compute alternate URLs via path substitution. URL patterns: EN = / (no language prefix) ZH-CN = /zh-cn/ JA = /ja/ */ -}} {{- $page := . -}} {{- $base := $.Site.BaseURL -}} {{- $urlPath := $page.RelPermalink -}} {{- if hasPrefix $urlPath "/" -}}{{ $urlPath = substr $urlPath 1 }}{{ end -}} {{- /* EN page: link to zh-cn and ja versions */ -}} {{- if eq .Site.Language.Lang "en" -}} {{- /* ZH-CN page: link to en and ja versions */ -}} {{- else if eq .Site.Language.Lang "zh-cn" -}} {{- $enPath := $urlPath -}} {{- if hasPrefix $urlPath "zh-cn/" }}{{ $enPath = substr $urlPath 6 }}{{ end -}} {{- /* JA page: link to en and zh-cn versions */ -}} {{- else if eq .Site.Language.Lang "ja" -}} {{- $enPath := $urlPath -}} {{- if hasPrefix $urlPath "ja/" }}{{ $enPath = substr $urlPath 3 }}{{ end -}} {{- end -}}
